docker -v //แสดงเวอร์ชั่น ของ docker
docker run hello-world //ทดสอบการทำงานของ docker ว่าสามารถรัน container ได้หรือไม่
docker ps //แสดง container ที่กำลังทำงานอยู่
docker ps --all //แสดง container ทั้งหมดที่มีอยู่ในเครื่อง ไม่ว่าจะกำลังทำงานหรือไม่ก็ตาม
docker run busybox ls //รัน container ที่ใช้ image ชื่อ busybox และสั่งให้แสดงไฟล์ใน container
docker run busybox ping google.com //รัน container ที่ใช้ image ชื่อ busybox และสั่งให้ ping google.com
docker stop <container_id> //หยุด container ที่กำลังทำงานอยู่ โดยใช้ container_id ที่ได้จากคำสั่ง docker ps
docker pull busybox //ดึง image ชื่อ busybox มาจาก Docker Hub
docker create busybox ping google.com //สร้าง container จาก image ชื่อ busybox และสั่งให้ ping google.com แต่ยังไม่เริ่มทำงาน
docker start <container_id> //เริ่มทำงาน container ที่สร้างไว้ โดยใช้ container_id ที่ได้จากคำสั่ง docker create
docker start -a <container_id> //เริ่มทำงาน container และแสดงผลลัพธ์ที่ได้จากการทำงานของ container
docker rm <container_id> //ลบ container ที่ไม่ต้องการ โดยใช้ container_id ที่ได้จากคำสั่ง docker ps --all
docker system prune //ลบ container, image, network และ volume ที่ไม่ได้ใช้งานแล้ว เพื่อเพิ่มพื้นที่ว่างในเครื่อง
docker run nginx //รัน container ที่ใช้ image ชื่อ nginx ซึ่งจะทำงานเป็น web server
docker exec <container_id> ls //รันคำสั่ง ls ใน container ที่กำลังทำงานอยู่ โดยใช้ container_id ที่ได้จากคำสั่ง docker ps
docker exec -it <container_id> sh //เข้าสู่ shell ของ container ที่กำลังทำงานอยู่ โดยใช้ container_id ที่ได้จากคำสั่ง docker ps
docker exec -it <container_id> bash //เข้าสู่ bash shell ของ container ที่กำลังทำงานอยู่ โดยใช้ container_id ที่ได้จากคำสั่ง docker ps (ถ้า image มี bash ให้ใช้งาน)
docker run -d -p 80:80 nginx //รัน container ที่ใช้ image ชื่อ nginx ในโหมด detached และเชื่อมต่อพอร์ต 80 ของ container กับพอร์ต 80 ของเครื่อง host
docker pull wordpress //ดึง image ชื่อ wordpress มาจาก Docker Hub
docker pull mysql:5.7 //ดึง image ชื่อ mysql เวอร์ชั่น 5.7 มาจาก Docker Hub
docker pull phpmyadmin/phpmyadmin //ดึง image ชื่อ phpmyadmin/phpmyadmin มาจาก Docker Hub

*Create network
docker network create wordpress //สร้าง network ชื่อ wordpress เพื่อให้ container ต่างๆ สามารถสื่อสารกันได้
docker network ls //แสดง network ทั้งหมดที่มีอยู่ในเครื่อง

docker run --name mysql --network wordpress -e MYSQL_ROOT_PASSWORD=111 -e MYSQL_DATABASE=wordpress_db -e MYSQL_USER=wp_user -e MYSQL_PASSWORD=111 -d mysql:5.7 //รัน container ชื่อ mysql ที่ใช้ image mysql:5.7 ในโหมด detached และเชื่อมต่อกับ network wordpress พร้อมตั้งค่าตัวแปรสภาพแวดล้อมสำหรับ MySQL
docker run --name pma --network wordpress -p 8888:80 -e PMA_ARBITRARY=1 -d phpmyadmin/phpmyadmin //รัน container ชื่อ pma ที่ใช้ image phpmyadmin/phpmyadmin ในโหมด detached และเชื่อมต่อกับ network wordpress พร้อมเชื่อมต่อพอร์ต 80 ของ container กับพอร์ต 8888 ของเครื่อง host
docker run -d --name wordpress --network wordpress -p 8080:80 -e WORDPRESS_DB_HOST=mysql -e WORDPRESS_DB_USER=wordpress -e WORDPRESS_DB_PASSWORD=wordpress -e WORDPRESS_DB_NAME=wordpress_db wordpress //รัน container ชื่อ wordpress ที่ใช้ image wordpress ในโหมด detached และเชื่อมต่อกับ network wordpress พร้อมเชื่อมต่อพอร์ต 80 ของ container กับพอร์ต 8080 ของเครื่อง host และตั้งค่าตัวแปรสภาพแวดล้อมสำหรับการเชื่อมต่อฐานข้อมูล MySQL
